cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build")
endif()

# Also covers compiler identification and compilers which are ccache symlinks.
# Build commands get the same settings through their compiler launchers below.
set(ENV{CCACHE_DIR} "${CMAKE_BINARY_DIR}/ccache")
set(ENV{CCACHE_TEMPDIR} "${CMAKE_BINARY_DIR}/ccache/tmp")
option(CQ_USE_CCACHE "Use ccache when available (cache stays in the build tree)" ON)
if(NOT CQ_USE_CCACHE)
    set(ENV{CCACHE_DISABLE} 1)
endif()
project(konsole_commandqueue VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Use a single-configuration generator such as Ninja or Unix Makefiles.")
endif()

set(cache_environment "${CMAKE_COMMAND};-E;env;CCACHE_DIR=${CMAKE_BINARY_DIR}/ccache;CCACHE_TEMPDIR=${CMAKE_BINARY_DIR}/ccache/tmp")
if(CQ_USE_CCACHE)
    find_program(CCACHE_EXECUTABLE ccache)
    if(CCACHE_EXECUTABLE)
        list(APPEND cache_environment "${CCACHE_EXECUTABLE}")
        message(STATUS "ccache enabled: ${CMAKE_BINARY_DIR}/ccache")
    else()
        message(STATUS "ccache not found; building without compiler caching")
    endif()
else()
    list(APPEND cache_environment "CCACHE_DISABLE=1")
    message(STATUS "ccache disabled")
endif()
set(CMAKE_C_COMPILER_LAUNCHER "${cache_environment}")
set(CMAKE_CXX_COMPILER_LAUNCHER "${cache_environment}")

find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH "${ECM_MODULE_PATH}")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
include(KDEInstallDirs6)
include(CTest)
include(cmake/Konsole.cmake)
add_subdirectory(plugin)
